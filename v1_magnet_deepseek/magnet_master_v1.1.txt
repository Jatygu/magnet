"""
logger.py — shared logging for MAGNET.
Writes JSONL events to memory/events.jsonl.
Used by all animals and utilities.
"""
import json
from pathlib import Path
from datetime import datetime, timezone

ROOT = Path(__file__).resolve().parent
MEMORY = ROOT / "memory"


def log_event(event, details=None, animal="system"):
    MEMORY.mkdir(parents=True, exist_ok=True)
    ts = datetime.now(timezone.utc).isoformat(timespec="seconds")
    record = {
        "time": ts,
        "animal": animal,
        "event": event,
        "details": details or {},
    }
    with open(MEMORY / "events.jsonl", "a", encoding="utf-8") as f:
        f.write(json.dumps(record) + "\n")
    return record


def load_config():
    import yaml
    cfg = ROOT / "config.yaml"
    if not cfg.exists():
        raise FileNotFoundError(f"config.yaml not found at {cfg}")
    with open(cfg, "r", encoding="utf-8") as f:
        return yaml.safe_load(f)

"""
llm_router.py — Below-ground LLM caller with fallback chain.
Calls Ollama. Tries each model in config.llm_models in order.
v1.1: Added keep_alive support to prevent model reload delays.
"""
import json
import requests
from logger import log_event, load_config


def call_ollama(prompt, system="", animal="unknown", max_tokens=None):
    config = load_config()
    models = config.get("llm_models") or ["qwen2.5:7b"]
    timeout = config.get("llm_timeout_sec", 360)
    host = config.get("ollama_host", "http://localhost:11434")
    temperature = config.get("temperature", 0.7)
    n_predict = max_tokens or config.get("max_tokens", 600)
    # v1.1 PATCH: Keep model in RAM between calls. Prevents reload delay +
    # Auditor timeouts on long overnight runs. Set in config.yaml.
    keep_alive = config.get("ollama_keep_alive", "60m")

    full_prompt = (system + "\n\n" + prompt) if system else prompt

    for i, model in enumerate(models):
        try:
            r = requests.post(
                f"{host}/api/generate",
                json={
                    "model": model,
                    "prompt": full_prompt,
                    "stream": False,
                    "keep_alive": keep_alive,
                    "options": {"temperature": temperature, "num_predict": n_predict},
                },
                timeout=timeout,
            )
            r.raise_for_status()
            text = r.json().get("response", "").strip()
            if text:
                log_event("llm_ok" if i == 0 else "llm_fallback_ok",
                          {"animal": animal, "model": model}, animal=animal)
                return text
            log_event("llm_empty", {"animal": animal, "model": model}, animal=animal)
        except requests.exceptions.Timeout:
            log_event("llm_timeout", {"animal": animal, "model": model}, animal=animal)
        except Exception as e:
            log_event("llm_error", {"animal": animal, "model": model,
                                    "error": str(e)[:120]}, animal=animal)
    log_event("llm_all_failed", {"animal": animal, "models": models}, animal=animal)
    return None


if __name__ == "__main__":
    print("LLM Router test...")
    print(call_ollama("Say 'MAGNET online' in 3 words.", animal="test"))

"""
scout.py — THE SCOUT (meerkat)
Quick reconnaissance. Tests whether sources are reachable.
Skips malformed entries gracefully.
"""
from pathlib import Path
import json
import requests
from logger import log_event, load_config

ROOT = Path(__file__).resolve().parent


def check_url(url):
    """Returns (status_code, size_bytes, error_string)."""
    try:
        r = requests.get(url, timeout=15,
                         headers={"User-Agent": "Mozilla/5.0 MAGNET/1.0"})
        return r.status_code, len(r.text), None
    except requests.exceptions.Timeout:
        return None, 0, "timeout"
    except requests.exceptions.ConnectionError as e:
        return None, 0, f"connection error: {str(e)[:80]}"
    except Exception as e:
        return None, 0, f"{type(e).__name__}: {str(e)[:80]}"


def main():
    print("\n=== SCOUT ===")
    try:
        config = load_config()
    except FileNotFoundError as e:
        print(f"  ! {e}")
        return

    raw_sources = config.get("sources") or []
    # Guard: skip anything that isn't a non-empty string
    sources = [s for s in raw_sources if isinstance(s, str) and s.strip()]
    skipped = len(raw_sources) - len(sources)
    if skipped > 0:
        print(f"  ! Skipped {skipped} malformed source entries")

    print(f"  Checking {len(sources)} sources...")
    notes = []
    for url in sources:
        status, size, err = check_url(url)
        ok = status is not None and 200 <= status < 400
        notes.append({"url": url, "status": status, "ok": ok,
                      "size": size, "error": err})
        if ok:
            print(f"    [OK]  {url}  HTTP {status}  ({size:,} bytes)")
        else:
            print(f"    [BAD] {url}  {err or f'HTTP {status}'}")

    notes_path = ROOT / "memory" / "scout_notes.jsonl"
    notes_path.parent.mkdir(parents=True, exist_ok=True)
    with open(notes_path, "w", encoding="utf-8") as f:
        for n in notes:
            f.write(json.dumps(n) + "\n")

    reachable = sum(1 for n in notes if n["ok"])
    log_event("scout_complete", {
        "sources": len(sources),
        "reachable": reachable,
        "skipped_malformed": skipped,
    }, animal="scout")
    print(f"  Summary: {reachable}/{len(sources)} reachable")
    print("  SCOUT OK")


if __name__ == "__main__":
    main()

"""
researcher.py — THE RESEARCHER (fox)
Scans allowlisted sources. Reddit via RSS; others via Playwright.
Extracts signals of unmet need. Writes memory/signals.jsonl.
v1.1: Reddit 403 retry + jittered delay to survive rate limiting.
"""
import json
import re
import time
import random
import requests
from pathlib import Path
from logger import log_event, load_config

ROOT = Path(__file__).resolve().parent
SIGNALS_FILE = ROOT / "memory" / "signals.jsonl"

KEYWORDS = [
    "need", "problem", "want", "struggle", "wish", "looking for",
    "trying to", "frustrated", "difficult", "hard to",
    "how do i", "how to", "is there a way", "is there a tool",
    "someone should", "i wish there", "would be nice if", "if only",
    "annoying", "frustrating", "broken", "doesn't work", "doesn't exist",
    "alternative to", "no good", "can't find", "missing", "gap in",
    "problem with", "issue with", "trouble with", "stuck on",
    "help me", "can't figure out", "why does", "why is it",
    "brokerage", "lead generation", "opportunity", "demand for",
    "willing to pay", "would pay for", "shut down", "discontinued",
    "expensive", "too costly", "affordable alternative",
]


def extract_signals(text):
    sentences = re.split(r"[.!?]\s+", text)
    out = []
    for s in sentences:
        s = s.strip()
        if len(s) < 20 or len(s) > 400:
            continue
        if any(k in s.lower() for k in KEYWORDS):
            out.append(s)
        if len(out) >= 15:
            break
    return out


def crawl_reddit_rss(url, _retry=False):
    """Crawl Reddit via RSS. Retries once on HTTP 403 (rate limit)."""
    rss = url.rstrip("/") + ".rss"
    headers = {
        "User-Agent": "Mozilla/5.0 MAGNET/1.0",
        "Accept": "application/rss+xml, application/xml, text/xml, */*",
    }
    try:
        r = requests.get(rss, timeout=20, headers=headers)
        # v1.1 PATCH: Retry once on 403 (Reddit rate-limit) after a longer pause
        if r.status_code == 403 and not _retry:
            log_event("researcher_reddit_403_retry",
                      {"url": url}, animal="researcher")
            time.sleep(30)
            return crawl_reddit_rss(url, _retry=True)
        r.raise_for_status()
        import xml.etree.ElementTree as ET
        root = ET.fromstring(r.text)
        ns = {"atom": "http://www.w3.org/2005/Atom"}
        entries = root.findall("atom:entry", ns)
        parts = []
        sub = url.split("/r/")[-1].rstrip("/") if "/r/" in url else "reddit"
        for e in entries[:25]:
            t = e.findtext("atom:title", "", ns)
            c = e.find("atom:content", ns)
            body = re.sub(r"<[^>]+>", "", c.text)[:500] if c is not None and c.text else ""
            if t:
                parts.append(t)
            if body:
                parts.append(body)
        text = "\n".join(parts)
        if not text:
            log_event("researcher_reddit_empty", {"url": url}, animal="researcher")
            return None
        return {"url": url, "title": f"Reddit r/{sub}",
                "text_excerpt": text[:1000],
                "signals": extract_signals(text)}
    except requests.exceptions.HTTPError as e:
        code = e.response.status_code if e.response is not None else "?"
        log_event("researcher_reddit_http_error",
                  {"url": url, "status": code, "retry": _retry,
                   "error": str(e)[:120]}, animal="researcher")
        return None
    except Exception as e:
        log_event("researcher_reddit_error",
                  {"url": url, "retry": _retry,
                   "error": f"{type(e).__name__}: {str(e)[:120]}"},
                  animal="researcher")
        return None


def crawl_playwright(url):
    try:
        from playwright.sync_api import sync_playwright
    except ImportError as e:
        log_event("researcher_playwright_missing",
                  {"error": str(e)}, animal="researcher")
        return None
    try:
        with sync_playwright() as p:
            b = p.chromium.launch(headless=True)
            pg = b.new_page()
            pg.goto(url, timeout=25000, wait_until="domcontentloaded")
            title = pg.title()
            text = pg.evaluate("() => document.body.innerText.substring(0, 3000)")
            b.close()
            return {"url": url, "title": title, "text_excerpt": text[:1000],
                    "signals": extract_signals(text)}
    except Exception as e:
        log_event("researcher_pw_error",
                  {"url": url,
                   "error_type": type(e).__name__,
                   "error": str(e)[:200]},
                  animal="researcher")
        return None


def main():
    print("\n=== RESEARCHER ===")
    try:
        config = load_config()
    except FileNotFoundError as e:
        print(f"  ! {e}")
        return

    raw_sources = config.get("sources") or []
    sources = [s for s in raw_sources if isinstance(s, str) and s.strip()]
    if len(sources) != len(raw_sources):
        print(f"  ! Skipped {len(raw_sources) - len(sources)} malformed sources")

    # v1.1 PATCH: Read delay from config (default 20s), add jitter to avoid rate-limit sync
    delay = config.get("reddit_delay_sec", 20)
    max_signals = config.get("max_signals_per_loop", 50)

    print(f"  {len(sources)} sources | delay {delay}s | max signals {max_signals}")
    all_signals = []
    errors = 0
    for url in sources:
        print(f"  Crawling: {url}")
        result = (crawl_reddit_rss(url) if "reddit.com" in url.lower()
                  else crawl_playwright(url))
        if result:
            print(f"    {result['title']} - {len(result['signals'])} signals")
            for s in result["signals"]:
                all_signals.append({"signal": s, "source": url,
                                    "title": result["title"]})
                log_event("researcher_found_signal",
                          {"signal": s[:200], "source": url},
                          animal="researcher")
        else:
            errors += 1
            print("    (no result)")
        # v1.1 PATCH: Jitter sleep so requests don't sync with Reddit's rate window
        jitter = random.uniform(0, delay * 0.5)
        time.sleep(delay + jitter)

    SIGNALS_FILE.parent.mkdir(parents=True, exist_ok=True)
    with open(SIGNALS_FILE, "a", encoding="utf-8") as f:
        for s in all_signals:
            f.write(json.dumps(s) + "\n")

    log_event("researcher_complete",
              {"sources": len(sources), "signals": len(all_signals),
               "errors": errors},
              animal="researcher")
    print(f"  Saved {len(all_signals)} signals ({errors} sources had no result)")
    print("  RESEARCHER OK")


if __name__ == "__main__":
    main()

"""
builder.py — THE BUILDER (beaver)
Takes recent signals, drafts helpful markdown resources to outbox/.
"""
import json
import re
from pathlib import Path
from llm_router import call_ollama
from logger import log_event, load_config

ROOT = Path(__file__).resolve().parent
SIGNALS_FILE = ROOT / "memory" / "signals.jsonl"
OUTBOX = ROOT / "outbox"


def sanitize(text, maxlen=50):
    t = re.sub(r"[^\w\s-]", "", text)[:maxlen].strip().replace(" ", "_").lower()
    return t or "untitled"


def title_from_draft(d):
    for line in d.split("\n"):
        if line.startswith("# "):
            return line[2:].strip()
    return "untitled"


def draft(signal, source, config):
    zoo = config.get("zoo_name", "magnet")
    purpose = config.get("purpose", "general")
    bm = config.get("business_model", "n/a")
    brand = config.get("branding_url", "Jatygu.com")
    prompt = f"""You are the Builder in MAGNET '{zoo}'.

Purpose: {purpose}
Business model: {bm}

The Researcher found this signal of an unmet need:

"{signal}"
(Source: {source})

Draft a SHORT, helpful, practical resource (max 400 words) that addresses
this need. It must be:
- Genuinely useful (not promotional)
- Plain-spoken (no marketing-speak, no hype)
- Markdown: a clear title, brief intro, 3-5 actionable steps
- Free to give away

Output ONLY markdown. No preamble.

Title format: # Helpful Resource: [descriptive title]
"""
    return call_ollama(prompt, system="You are a plain-spoken, helpful writer.",
                       animal="builder")


def main():
    print("\n=== BUILDER ===")
    try:
        config = load_config()
    except FileNotFoundError as e:
        print(f"  ! {e}")
        return

    if not SIGNALS_FILE.exists():
        print("  No signals yet. Run researcher.py first.")
        return

    signals = []
    with open(SIGNALS_FILE, "r", encoding="utf-8") as f:
        for line in f:
            if line.strip():
                signals.append(json.loads(line))
    if not signals:
        print("  Signal file empty.")
        return

    max_drafts = config.get("max_drafts_per_loop", 3)
    target = signals[-max_drafts:][::-1]
    print(f"  Drafting {len(target)} of {len(signals)} signals...")

    OUTBOX.mkdir(parents=True, exist_ok=True)
    brand = config.get("branding_url", "Jatygu.com")
    for i, item in enumerate(target, 1):
        sig = item["signal"]
        src = item["source"]
        print(f"  [{i}] {sig[:80]}...")
        text = draft(sig, src, config)
        if not text:
            log_event("builder_skipped", {"reason": "LLM returned nothing"},
                      animal="builder")
            continue
        title = title_from_draft(text)

def keyword_audit(path, text):
    banned = ["send email", "dm me", "click here to buy", "act now",

def audit(path, const, config):
    text = path.read_text(encoding="utf-8")
    prompt = f"""You are the Auditor in MAGNET '{config.get('zoo_name')}'.

CONSTITUTION (excerpt):
{const[:1500]}

DRAFT:
{text[:2000]}

Check:
1. Sends email/DM/social post? (MUST NOT)
2. Spends money? (MUST NOT)
3. Uses credentials? (MUST NOT)
4. Unsolicited sales offer? (SHOULD NOT)

7. Factual claims verifiable? (MUST)

    resp = call_ollama(prompt, system="Strict but fair auditor. JSON only.",
                       animal="auditor", max_tokens=400)

        except json.JSONDecodeError:
            pass
    return keyword_audit(path, text)

    path.write_text(text, encoding="utf-8")

def run_animal(name):
    script = ROOT / f"{name}.py"
    if not script.exists():
        print(f"  ! {name}.py missing")
        return False
    t0 = time.time()
    # v1.1 PATCH: Force UTF-8 stdout in child processes.
    # Without this, Windows PowerShell cp1252 crashes when an animal
    # prints an emoji or non-Latin character. Fixed 3 crashes on 2026-09-29.
    env = os.environ.copy()
    env["PYTHONIOENCODING"] = "utf-8"
    env["PYTHONUTF8"] = "1"
    try:
        proc = subprocess.run(
            [sys.executable, str(script)],
            cwd=str(ROOT),
            capture_output=True,
            text=True,
            timeout=1200,
            encoding="utf-8",
            errors="replace",
            env=env,
        )
        rt = time.time() - t0
        status = "OK" if proc.returncode == 0 else "FAIL"
        print(f"  [{name:11s}] {status} ({rt:.1f}s)")
        if proc.stdout:
            for line in proc.stdout.split("\n")[-6:]:
                if line.strip():
                    print(f"    {line}")
        if proc.returncode != 0 and proc.stderr:
            print(f"    ERR: {proc.stderr[-300:]}")
        return proc.returncode == 0
    except subprocess.TimeoutExpired:
        print(f"  [{name}] TIMEOUT")
        return False
    except Exception as e:
        print(f"  [{name}] EXCEPTION: {e}")
        return False

"""extract.py — MAGNET envelope extractor (flat mode)."""
import sys
import re
from pathlib import Path

MARKER = re.compile(r"^=== FILE: (.+?) ===$", re.MULTILINE)


def extract(master_path, dest=None):
    master = Path(master_path).resolve()
    if not master.exists():
        print(f"  ! Not found: {master}")
        sys.exit(1)
    content = master.read_text(encoding="utf-8")
    base = Path(dest).resolve() if dest else master.parent

    matches = list(MARKER.finditer(content))
    if not matches:
        print(f"  ! No markers found in {master}")
        sys.exit(1)

    print()
    print("=" * 60)
    print("  MAGNET Extract (flat mode)")
    print("=" * 60)
    print(f"  Master: {master}")
    print(f"  Dest:   {base}")
    print(f"  Files:  {len(matches)}")
    print()

    ok = 0
    for i, m in enumerate(matches):
        path = m.group(1).strip()
        start = m.end()
        nl = content.find("\n", start)
        if nl == -1:
            continue
        start = nl + 1
        end = matches[i + 1].start() if i + 1 < len(matches) else len(content)
        raw = content[start:end]
        raw = re.sub(r"\n=+\s*$", "", raw)
        raw = raw.strip() + "\n"
        flat = Path(path).name
        out = base / flat
        try:
            out.parent.mkdir(parents=True, exist_ok=True)

"""
pack.py — repacks the flat MAGNET folder into a single master envelope.
Run this after editing any file, to keep magnet_master_v1.1.txt current.
"""
from pathlib import Path

ROOT = Path(__file__).resolve().parent
OUT = ROOT / "magnet_master_v1.1.txt"

FILES = [
    "constitution.txt", "README.md", "config.yaml.template",
    "logger.py", "llm_router.py", "tether.py", "dashboard.py",
    "coordinator.py", "scout.py", "researcher.py", "builder.py",
    "auditor.py", "publisher.py", "tracker.py", "historian.py",
    "ledger.py", "setup.py", "orchestrator.py", "extract.py",
    "pack.py", "gist.py",
]


def main():
    print(f"Packing {len(FILES)} files into {OUT.name}")
    chunks = []
    chunks.append("=== FILE: magnet_master_v1.1.txt ===")
    chunks.append("MAGNET v1.1 - Master Envelope")
    chunks.append("Regenerate with: python pack.py")
    chunks.append("")
    for name in FILES:
        p = ROOT / name
        if not p.exists():
            print(f"  ! missing: {name}")
            continue
        print(f"  + {name}")
        chunks.append(f"=== FILE: {name} ===")
        chunks.append(p.read_text(encoding="utf-8"))
    chunks.append("=== END ===")
    OUT.write_text("\n".join(chunks), encoding="utf-8")
    print(f"\nWrote {OUT} ({OUT.stat().st_size:,} bytes)")


if __name__ == "__main__":
    main()

"""
gist.py — publishes magnet_master_v1.1.txt to a GitHub Gist.
Requires GITHUB_TOKEN in environment or .env file.
"""
import os
import requests
from pathlib import Path

ROOT = Path(__file__).resolve().parent
MASTER = ROOT / "magnet_master_v1.1.txt"

try:
    from dotenv import load_dotenv
    load_dotenv()
except ImportError:
    pass


def publish():
    token = os.environ.get("GITHUB_TOKEN")
    if not token:
        print("  ! GITHUB_TOKEN not set. Create a token at:")
        print("    https://github.com/settings/tokens (scope: gist)")
        print("  Then create .env with: GITHUB_TOKEN=xxx")
        return
    if not MASTER.exists():
        print(f"  ! {MASTER} not found. Run pack.py first.")
        return

    content = MASTER.read_text(encoding="utf-8")
    payload = {
        "description": "MAGNET v1.1 master envelope",
        "public": False,
        "files": {"magnet_master_v1.1.txt": {"content": content}},
    }
    r = requests.post(
        "https://api.github.com/gists",
        headers={"Authorization": f"token {token}",
                 "Accept": "application/vnd.github+json"},
        json=payload, timeout=30,
    )
    if r.status_code not in (200, 201):
        print(f"  ! GitHub API error {r.status_code}: {r.text[:300]}")
        return
    data = r.json()
    raw_url = data["files"]["magnet_master_v1.1.txt"]["raw_url"]
    html_url = data["html_url"]
    print("  + Gist created")
    print(f"    HTML: {html_url}")
    print(f"    Raw:  {raw_url}")


if __name__ == "__main__":
    publish()
